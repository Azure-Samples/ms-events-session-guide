{{- /* Normalizes and validates one event bundle's resources (catalog.json and guides.json) into the payload the event guide loads at runtime. Context: the event bundle page. Use partialCached keyed by the event ID. Raw bundle resources are read as content only: never request their permalinks, which would publish the source files. Returns dict "event" (payload metadata) "manifestEntry" "resource" (fingerprinted). */ -}} {{- $page := . -}} {{- $meta := partialCached "events/metadata.html" $page $page.File.ContentBaseName -}} {{- $where := printf "content/events/%s" $meta.id -}} {{- $read := dict -}} {{- range $name := slice "catalog.json" "guides.json" -}} {{- with $page.Resources.Get $name -}} {{- with try (transform.Unmarshal .Content) -}} {{- if .Err -}}{{- errorf "%s/%s is not valid JSON: %s" $where $name .Err -}}{{- else -}}{{- $read = merge $read (dict $name .Value) -}}{{- end -}} {{- end -}} {{- else -}} {{- errorf "%s/%s is missing" $where $name -}} {{- end -}} {{- end -}} {{- $normalized := partial (printf "events/format-%s.html" $meta.catalogFormat) (dict "source" (index $read "catalog.json") "where" (printf "%s/catalog.json" $where)) -}} {{- $codes := partial "events/instances.html" (dict "instances" $normalized.instances "where" (printf "%s/catalog.json" $where)) -}} {{- $guide := partial "events/guides.html" (dict "guides" (index $read "guides.json") "codes" $codes "where" (printf "%s/guides.json" $where)) -}} {{- $content := dict "instances" $normalized.instances "guide" $guide "updated" $normalized.updated -}} {{- $catalogVersion := substr (jsonify $content | sha256) 0 16 -}} {{- $event := dict "id" $meta.id "name" $meta.name "summary" $meta.summary "startDate" $meta.startDate "endDate" $meta.endDate "timeZone" $meta.timeZone "catalogUrl" $meta.catalogUrl "url" $meta.url "catalogVersion" $catalogVersion -}} {{- with $meta.registrationUrl -}} {{- $event = merge $event (dict "registrationUrl" .) -}} {{- end -}} {{- $payload := merge $content (dict "event" $event) -}} {{- $resource := resources.FromString (printf "data/events/%s/catalog.json" $meta.id) (jsonify $payload) | fingerprint "sha256" -}} {{- $manifestEntry := merge $event (dict "catalogAssetUrl" $resource.RelPermalink "keywords" $meta.keywords) -}} {{- return dict "event" $event "meta" $meta "manifestEntry" $manifestEntry "resource" $resource "guide" $guide -}}
