{{- /* Wraps the telemetry configuration partial and derives the js.Build defines. Returns dict "enabled" "config" "defines". Disabled when the configuration partial is absent or reports disabled, so builds emit no analytics code. */ -}} {{- $state := dict "enabled" false "config" nil -}} {{- if templates.Exists "partials/telemetry-config.html" -}} {{- $state = partialCached "telemetry-config.html" . "telemetry" -}} {{- end -}} {{- $enabled := and $state.enabled $state.config -}} {{- return dict "enabled" (cond $enabled true false) "config" (cond $enabled $state.config nil) "defines" (dict "__TELEMETRY_ENABLED__" (cond $enabled "true" "false")) -}}
